def median(numbers):
    """Return the median of a non-empty list of numbers."""
    if not numbers:
        raise ValueError("median() requires a non-empty list")

    ordered = sorted(numbers)
    n = len(ordered)
    mid = n // 2

    if n % 2 == 1:
        return ordered[mid]

    return (ordered[mid - 1] + ordered[mid]) / 2
