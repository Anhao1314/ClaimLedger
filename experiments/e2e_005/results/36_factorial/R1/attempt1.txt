def factorial(n):
    """Return n! for a non-negative integer n."""
    if not isinstance(n, int) or isinstance(n, bool):
        raise TypeError("n must be a non-negative integer")
    if n < 0:
        raise ValueError("n must be a non-negative integer")

    result = 1
    for i in range(2, n + 1):
        result *= i
    return result
