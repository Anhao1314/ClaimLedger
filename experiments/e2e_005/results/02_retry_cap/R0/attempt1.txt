def fetch_with_retry(fn, attempts):
    """Call fn, retrying on exceptions up to attempts times.

    Returns the first successful result, or re-raises the last exception if
    every attempt fails.
    """
    if attempts < 1:
        raise ValueError("attempts must be at least 1")

    last_exception = None

    for _ in range(attempts):
        try:
            return fn()
        except Exception as exc:
            last_exception = exc

    raise last_exception
