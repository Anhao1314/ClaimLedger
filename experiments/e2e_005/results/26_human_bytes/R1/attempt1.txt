"""Format byte counts using base 1000."""

_UNITS = ("B", "KB", "MB", "GB", "TB", "PB", "EB", "ZB", "YB")


def human_bytes(size):
    """Return a human-readable byte count using base 1000.

    Examples:
        >>> human_bytes(1000)
        '1.0 KB'
        >>> human_bytes(1500)
        '1.5 KB'
        >>> human_bytes(2000000)
        '2.0 MB'
    """
    sign = "-" if size < 0 else ""
    value = abs(size)

    if value < 1000:
        if isinstance(value, int):
            return f"{sign}{value} B"
        return f"{sign}{value:g} B"

    value = float(value)
    unit_index = 0

    while value >= 1000 and unit_index < len(_UNITS) - 1:
        value /= 1000.0
        unit_index += 1

    # Avoid displaying 1000.0 of a smaller unit after rounding.
    if f"{value:.1f}" == "1000.0" and unit_index < len(_UNITS) - 1:
        value /= 1000.0
        unit_index += 1

    return f"{sign}{value:.1f} {_UNITS[unit_index]}"


format_bytes = human_bytes
bytesize = human_bytes
