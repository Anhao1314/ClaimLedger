def chunks(sequence, size):
    """Split a sequence into consecutive lists of at most `size` items."""
    if size <= 0:
        raise ValueError("size must be greater than 0")

    iterator = iter(sequence)
    result = []

    while True:
        batch = []
        try:
            for _ in range(size):
                batch.append(next(iterator))
        except StopIteration:
            if batch:
                result.append(batch)
            break
        result.append(batch)

    return result


chunk = chunks
