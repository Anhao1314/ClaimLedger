def median(numbers):
    """Return the median of a non-empty list of numbers."""
    if not numbers:
        raise ValueError("median() requires a non-empty list")

    values = sorted(numbers)
    count = len(values)
    middle = count // 2

    if count % 2 == 1:
        return values[middle]

    return (values[middle - 1] + values[middle]) / 2
