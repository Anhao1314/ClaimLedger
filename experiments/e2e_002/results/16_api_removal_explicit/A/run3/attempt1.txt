"""Public API.

The legacy first-generation endpoints have been retired. Only version 2 of
the API is exposed.
"""

from typing import Dict, List

from fastapi import FastAPI, HTTPException, status
from pydantic import BaseModel

app = FastAPI(title="Public API", version="2.0.0")


class Item(BaseModel):
    id: int
    name: str


_items: Dict[int, Item] = {}


@app.get("/v2/health")
def health() -> dict:
    return {"status": "ok", "api_version": "2"}


@app.get("/v2/items", response_model=List[Item])
def list_items() -> List[Item]:
    return list(_items.values())


@app.post("/v2/items", response_model=Item, status_code=status.HTTP_201_CREATED)
def create_item(item: Item) -> Item:
    if item.id in _items:
        raise HTTPException(
            status_code=status.HTTP_409_CONFLICT,
            detail="Item already exists",
        )
    _items[item.id] = item
    return item
