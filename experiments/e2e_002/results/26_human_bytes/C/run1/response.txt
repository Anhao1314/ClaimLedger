"""Format byte counts in a human-readable form (base 1000)."""

_UNITS = ("B", "KB", "MB", "GB", "TB", "PB", "EB", "ZB", "YB")


def human_bytes(size, precision=1):
    """Return *size* formatted with base-1000 units.

    Examples:
        1000 -> "1.0 KB"
        1500 -> "1.5 KB"
        2000000 -> "2.0 MB"
    """
    if isinstance(size, bool):
        size = int(size)
    if not isinstance(size, (int, float)):
        raise TypeError("size must be a number")
    if size != size or size in (float("inf"), float("-inf")):
        raise ValueError("size must be finite")

    sign = "-" if size < 0 else ""
    value = abs(float(size))
    unit_index = 0

    while value >= 1000 and unit_index < len(_UNITS) - 1:
        value /= 1000
        unit_index += 1

    if unit_index == 0:
        if value.is_integer():
            return f"{sign}{int(value)} B"
        return f"{sign}{value:g} B"

    return f"{sign}{value:.{precision}f} {_UNITS[unit_index]}"


format_bytes = human_bytes
bytesize = human_bytes
humanize_bytes = human_bytes
human_readable_bytes = human_bytes
human_readable = human_bytes
sizeof_fmt = human_bytes
