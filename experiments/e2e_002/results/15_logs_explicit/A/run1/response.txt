"""Structured logging setup for the service."""

from __future__ import annotations

import contextvars
import json
import logging
import os
import sys
from contextlib import contextmanager
from datetime import datetime, timezone
from typing import Any, Dict, Iterator, Optional, Union

__all__ = [
    "JsonFormatter",
    "bind_context",
    "clear_context",
    "configure_logging",
    "get_context",
    "get_logger",
    "log_context",
    "reset_context",
    "setup_logging",
]

_LOG_CONTEXT = contextvars.ContextVar("log_context", default={})

_RESERVED_LOG_RECORD_ATTRS = {
    "args",
    "asctime",
    "created",
    "exc_info",
    "exc_text",
    "filename",
    "funcName",
    "levelname",
    "levelno",
    "lineno",
    "message",
    "module",
    "msecs",
    "msg",
    "name",
    "pathname",
    "process",
    "processName",
    "relativeCreated",
    "stack_info",
    "thread",
    "threadName",
    "taskName",
}


def get_context() -> Dict[str, Any]:
    """Return a copy of the current logging context."""
    return dict(_LOG_CONTEXT.get())


def bind_context(**kwargs: Any) -> contextvars.Token:
    """Add values to the current logging context."""
    context = get_context()
    context.update(kwargs)
    return _LOG_CONTEXT.set(context)


def reset_context(token: contextvars.Token) -> None:
    """Restore a previous logging context."""
    _LOG_CONTEXT.reset(token)


def clear_context() -> None:
    """Clear all values from the current logging context."""
    _LOG_CONTEXT.set({})


@contextmanager
def log_context(**kwargs: Any) -> Iterator[None]:
    """Temporarily add values to the logging context."""
    token = bind_context(**kwargs)
    try:
        yield
    finally:
        reset_context(token)


class JsonFormatter(logging.Formatter):
    """Format log records as single-line JSON objects."""

    def __init__(
        self,
        service_name: str = "service",
        environment: str = "development",
        **kwargs: Any,
    ) -> None:
        super().__init__(**kwargs)
        self.service_name = service_name
        self.environment = environment

    def format(self, record: logging.LogRecord) -> str:
        payload: Dict[str, Any] = {
            "timestamp": datetime.fromtimestamp(
                record.created, tz=timezone.utc
            ).isoformat(),
            "level": record.levelname,
            "logger": record.name,
            "message": record.getMessage(),
            "service": self.service_name,
            "environment": self.environment,
            "module": record.module,
            "function": record.funcName,
            "line": record.lineno,
            "process": record.process,
            "thread": record.threadName,
            "context": get_context(),
        }

        for key, value in record.__dict__.items():
            if key in _RESERVED_LOG_RECORD_ATTRS or key.startswith("_"):
                continue
            if key not in payload:
                payload[key] = value

        if record.exc_info:
            payload["exception"] = self.formatException(record.exc_info)

        if record.stack_info:
            payload["stack"] = self.formatStack(record.stack_info)

        return json.dumps(payload, default=str, ensure_ascii=False)


def configure_logging(
    service_name: Optional[str] = None,
    level: Union[str, int, None] = None,
    environment: Optional[str] = None,
    *,
    json_logs: Optional[bool] = None,
    stream: Optional[Any] = None,
) -> logging.Logger:
    """Configure root logging for the service and return the root logger."""
    service_name = service_name or os.getenv("SERVICE_NAME", "service")
    environment = (
        environment
        or os.getenv("ENVIRONMENT")
        or os.getenv("ENV")
        or "development"
    )

    if level is None:
        level = os.getenv("LOG_LEVEL", "INFO")

    if isinstance(level, str):
        level = getattr(logging, level.upper(), logging.INFO)

    if json_logs is None:
        log_format = os.getenv("LOG_FORMAT", "json").strip().lower()
        json_logs = log_format not in {"text", "plain", "console"}

    if json_logs:
        formatter = JsonFormatter(
            service_name=service_name,
            environment=environment,
        )
    else:
        formatter = logging.Formatter(
            "%(asctime)s %(levelname)s %(name)s %(message)s"
        )

    handler = logging.StreamHandler(stream or sys.stdout)
    handler.setFormatter(formatter)
    setattr(handler, "_structured_logging_handler", True)

    root = logging.getLogger()

    for existing_handler in list(root.handlers):
        if getattr(existing_handler, "_structured_logging_handler", False):
            root.removeHandler(existing_handler)

    root.addHandler(handler)
    root.setLevel(level)

    return root


setup_logging = configure_logging


def get_logger(name: Optional[str] = None) -> logging.Logger:
    """Return a named logger for the application."""
    return logging.getLogger(name)
