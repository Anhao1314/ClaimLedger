def _current_context() -> dict[str, Any]:
    """Return a mutable copy of the current logging context."""

    return dict(_CONTEXT.get() or {})


def bind_context(**values: Any) -> Token:
    """Attach values to the current async/thread context.

    The returned token can be used with ``_CONTEXT.reset(token)``. Most code
    should prefer :func:`log_context` instead of managing tokens directly.
    """

    context = _current_context()
    context.update(values)
    return _CONTEXT.set(context)


def unbind_context(*keys: str) -> None:
    """Remove specific keys from the current logging context.

    If no keys are supplied, the entire context is cleared.
    """

    if not keys:
        clear_context()
        return

    context = _current_context()
    for key in keys:
        context.pop(key, None)
    _CONTEXT.set(context)


def clear_context() -> None:
    """Remove all values from the current logging context."""

    _CONTEXT.set({})
