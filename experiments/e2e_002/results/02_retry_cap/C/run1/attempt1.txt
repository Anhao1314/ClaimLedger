RETRY_CAP = 3


def fetch_with_retry(fn, attempts):
    if attempts < 1:
        raise ValueError("attempts must be at least 1")

    max_attempts = min(attempts, RETRY_CAP)
    last_exception = None

    for _ in range(max_attempts):
        try:
            return fn()
        except Exception as exc:
            last_exception = exc

    raise last_exception
