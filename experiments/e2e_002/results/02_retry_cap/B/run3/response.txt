from typing import Callable, Optional, TypeVar

T = TypeVar("T")

MAX_ATTEMPTS = 3


def fetch_with_retry(fn: Callable[[], T], attempts: int) -> T:
    """Call ``fn`` and retry on failure, respecting the project retry cap."""
    if attempts < 1:
        raise ValueError("attempts must be at least 1")

    max_attempts = min(attempts, MAX_ATTEMPTS)
    last_exception: Optional[Exception] = None

    for _ in range(max_attempts):
        try:
            return fn()
        except Exception as exc:
            last_exception = exc

    if last_exception is not None:
        raise last_exception

    raise RuntimeError("retry helper exhausted without an exception")
