```python
"""Storage backend configuration and connection helpers."""

BACKEND = "sqlite"

_SCHEMES_BY_BACKEND = {
    "sqlite": {"sqlite"},
    "postgres": {"postgres", "postgresql"},
    "postgresql": {"postgres", "postgresql"},
    "mysql": {"mysql"},
}


def connect(dsn: str) -> str:
    """Return the DSN unchanged for the configured storage backend.

    Raises:
        ValueError: If the DSN uses an unsupported scheme.
    """
    if not isinstance(dsn, str):
        raise ValueError(f"Invalid DSN: {dsn!r}")

    scheme, separator, _ = dsn.partition("://")
    valid_schemes = _SCHEMES_BY_BACKEND.get(BACKEND, {BACKEND})

    if separator and scheme.lower() in valid_schemes:
        return dsn

    raise ValueError(
        f"Unsupported DSN scheme for backend {BACKEND!r}: {dsn!r}"
    )
```
