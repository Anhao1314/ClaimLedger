```python
"""Parse ISO-formatted calendar dates."""

from __future__ import annotations

import re
from datetime import date

_ISO_DATE_PATTERN = re.compile(r"^\d{4}-\d{2}-\d{2}$")


def parse_date(s: str) -> date:
    """Parse a date string in strict YYYY-MM-DD form.

    Raises:
        ValueError: If *s* is not a valid YYYY-MM-DD date.
    """
    if not isinstance(s, str) or _ISO_DATE_PATTERN.fullmatch(s) is None:
        raise ValueError("Expected a date in YYYY-MM-DD format")

    try:
        return date.fromisoformat(s)
    except ValueError as exc:
        raise ValueError(f"Invalid date: {s!r}") from exc
```
