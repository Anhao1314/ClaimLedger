```python
import time

# No project-wide retry cap is stated in MEMORY, so the caller-provided
# attempts value is the effective cap.
_RETRY_CAP = None

_BACKOFF_BASE_SECONDS = 0.001
_BACKOFF_MAX_SECONDS = 0.01


def fetch_with_retry(fn, attempts):
    if attempts < 1:
        raise ValueError("attempts must be at least 1")

    max_attempts = attempts if _RETRY_CAP is None else min(attempts, _RETRY_CAP)
    last_exc = None

    for attempt in range(max_attempts):
        try:
            return fn()
        except Exception as exc:
            last_exc = exc
            if attempt + 1 >= max_attempts:
                raise

            delay = min(
                _BACKOFF_BASE_SECONDS * (2 ** min(attempt, 10)),
                _BACKOFF_MAX_SECONDS,
            )
            time.sleep(delay)

    raise last_exc
```
