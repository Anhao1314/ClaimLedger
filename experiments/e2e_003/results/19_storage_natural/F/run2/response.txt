"""Analytics storage layer.

The durable database backend is intentionally not selected here because the
repository's database constraints are pending human review. This module
provides a storage boundary that always redacts PII before any backend write,
plus a thread-safe in-memory backend for local use and tests.
"""

from __future__ import annotations

import copy
import re
import threading
from collections.abc import Mapping, Sequence
from typing import Any, Optional, Protocol

REDACTED = "[REDACTED]"

__all__ = [
    "AnalyticsStorage",
    "AnalyticsStorageService",
    "StorageBackend",
    "REDACTED",
    "redact_pii",
]

_PII_KEY_PATTERN = re.compile(
    r"(?i)(?:email|e[-_]?mail|phone|mobile|telephone|ssn|social[-_ ]?security|"
    r"credit[-_ ]?card|card[-_ ]?number|address|first[-_ ]?name|last[-_ ]?name|"
    r"full[-_ ]?name|ip[-_ ]?address|date[-_ ]?of[-_ ]?birth|dob|passport|"
    r"driver[-_ ]?license|license[-_ ]?number)"
)

_EMAIL_RE = re.compile(r"(?i)\b[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,}\b")
_SSN_RE = re.compile(r"(?<!\d)\d{3}-\d{2}-\d{4}(?!\d)")
_IPV4_RE = re.compile(r"(?<!\d)(?:\d{1,3}\.){3}\d{1,3}(?!\d)")
_CREDIT_CARD_RE = re.compile(r"(?<!\d)(?:\d[ -]*?){13,19}(?!\d)")
_PHONE_RE = re.compile(
    r"(?<!\d)(?:\+?1[-.\s]?)?(?:\(?\d{3}\)?[-.\s]?)\d{3}[-.\s]?\d{4}(?!\d)"
)


def _is_pii_key(key: Any) -> bool:
    if not isinstance(key, str):
        return False
    key = key.strip()
    return key.lower() == "name" or bool(_PII_KEY_PATTERN.search(key))


def _redact_string(value: str) -> str:
    value = _EMAIL_RE.sub(REDACTED, value)
    value = _SSN_RE.sub(REDACTED, value)
    value = _IPV4_RE.sub(REDACTED, value)
    value = _CREDIT_CARD_RE.sub(REDACTED, value)
    value = _PHONE_RE.sub(REDACTED, value)
    return value


def redact_pii(value: Any) -> Any:
    """Return a copy of ``value`` with PII redacted.

    Redaction is recursive and never mutates the caller's object. It is applied
    at the storage boundary before data is written to any backend.
    """
    if isinstance(value, Mapping):
        redacted: dict[Any, Any] = {}
        for key, item in value.items():
            redacted[key] = REDACTED if _is_pii_key(key) else redact_pii(item)
        return redacted

    if isinstance(value, list):
        return [redact_pii(item) for item in value]

    if isinstance(value, tuple):
        return tuple(redact_pii(item) for item in value)

    if isinstance(value, set):
        return {redact_pii(item) for item in value}

    if isinstance(value, str):
        return _redact_string(value)

    return copy.deepcopy(value)


class StorageBackend(Protocol):
    def append(self, event: Mapping[str, Any]) -> None:
        ...

    def all(self) -> Sequence[Mapping[str, Any]]:
        ...

    def clear(self) -> None:
        ...


class AnalyticsStorage:
    """Thread-safe storage for analytics events with mandatory PII redaction."""

    def __init__(self) -> None:
        self._events: list[dict[str, Any]] = []
        self._lock = threading.RLock()

    def append(self, event: Mapping[str, Any]) -> None:
        if not isinstance(event, Mapping):
            raise TypeError("event must be a mapping")

        safe_event = redact_pii(dict(event))
        with self._lock:
            self._events.append(safe_event)

    def store(self, event: Mapping[str, Any]) -> None:
        self.append(event)

    def record(self, event: Mapping[str, Any]) -> None:
        self.append(event)

    def all(self) -> list[dict[str, Any]]:
        with self._lock:
            return copy.deepcopy(self._events)

    def clear(self) -> None:
        with self._lock:
            self._events.clear()

    def __len__(self) -> int:
        with self._lock:
            return len(self._events)


class AnalyticsStorageService:
    """Service wrapper around a storage backend.

    The wrapper redacts PII before handing data to the backend, and compliant
    backends must redact again at their own write boundary.
    """

    def __init__(self, backend: Optional[StorageBackend] = None) -> None:
        self._backend = backend or AnalyticsStorage()

    def record(self, event: Mapping[str, Any]) -> None:
        if not isinstance(event, Mapping):
            raise TypeError("event must be a mapping")
        self._backend.append(redact_pii(dict(event)))

    def events(self) -> list[dict[str, Any]]:
        return [dict(event) for event in self._backend.all()]

    def clear(self) -> None:
        self._backend.clear()
