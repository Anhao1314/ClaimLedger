```python
"""PostgreSQL-backed storage for the analytics service.

Configuration is read exclusively from environment variables:

    ANALYTICS_DATABASE_URL   preferred PostgreSQL DSN
    DATABASE_URL             fallback PostgreSQL DSN

User-facing text is not produced in this layer. Failures carry stable
message keys (``error.analytics.*``) that callers can externalize and localize.
"""

from __future__ import annotations

import base64
import json
import os
from collections.abc import Mapping
from dataclasses import dataclass
from datetime import datetime, timezone
from typing import Any, Optional

import psycopg2
from psycopg2.extras import Json, RealDictCursor

DEFAULT_PAGE_SIZE = 20
MAX_PAGE_SIZE = 100


class StorageError(Exception):
    """Storage failure identified by an externalized message key."""

    def __init__(self, code: str) -> None:
        super().__init__(code)
        self.code = code


@dataclass(frozen=True)
class Page:
    items: list[dict[str, Any]]
    next_cursor: Optional[str]


def _database_url() -> str:
    url = os.environ.get("ANALYTICS_DATABASE_URL") or os.environ.get("DATABASE_URL")
    if not url:
        raise StorageError("error.analytics.database_not_configured")
    return url


def _as_utc(value: datetime) -> datetime:
    if not isinstance(value, datetime):
        raise StorageError("error.analytics.invalid_timestamp")
    if value.tzinfo is None:
        return value.replace(tzinfo=timezone.utc)
    return value.astimezone(timezone.utc)


def _encode_cursor(occurred_at: datetime, event_id: int) -> str:
    payload = {
        "t": _as_utc(occurred_at).isoformat(),
        "i": int(event_id),
    }
    raw = json.dumps(payload, separators=(",", ":")).encode("utf-8")
    return base64.urlsafe_b64encode(raw).decode("ascii").rstrip("=")


def _decode_cursor(cursor: str) -> tuple[datetime, int]:
    try:
        padded = cursor + ("=" * (-len(cursor) % 4))
        raw = base64.urlsafe_b64decode(padded.encode("ascii"))
        payload = json.loads(raw.decode("utf-8"))
        occurred_at = datetime.fromisoformat(payload["t"])
        event_id = int(payload["i"])
    except Exception:
        raise StorageError("error.analytics.invalid_cursor") from None

    if occurred_at.tzinfo is None:
        occurred_at = occurred_at.replace(tzinfo=timezone.utc)
    return occurred_at.astimezone(timezone.utc), event_id


class AnalyticsStorage:
    """Analytics event storage backed by PostgreSQL."""

    def __init__(self) -> None:
        self._dsn = _database_url()
        self._conn: Optional[psycopg2.extensions.connection] = None
        self._schema_ready = False

    def _connection(self) -> psycopg2.extensions.connection:
        if self._conn is not None and not self._conn.closed:
            return self._conn

        try:
            conn = psycopg2.connect(self._dsn)
        except psycopg2.Error as exc:
            raise StorageError("error.analytics.database_unavailable") from exc

        self._conn = conn
        try:
            with conn.cursor() as cur:
                cur.execute("SET TIME ZONE 'UTC'")
            conn.commit()
            self._ensure_schema(conn)
        except Exception:
            conn.close()
            self._conn = None
            raise

        return conn

    def _ensure_schema(self, conn: psycopg2.extensions.connection) -> None:
        if self._schema_ready:
            return

        statements = (
            """
            CREATE TABLE IF NOT EXISTS analytics_events (
                id BIGSERIAL PRIMARY KEY,
                name TEXT NOT NULL,
                user_id TEXT,
                session_id TEXT,
                properties JSONB NOT NULL DEFAULT '{}'::jsonb,
                occurred_at TIMESTAMPTZ NOT NULL,
                received_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
            )
            """,
            """
            CREATE INDEX IF NOT EXISTS analytics_events_order_idx
                ON analytics_events (occurred_at DESC, id DESC)
            """,
            """
            CREATE INDEX IF NOT EXISTS analytics_events_name_order_idx
                ON analytics_events (name, occurred_at DESC, id DESC)
            """,
            """
            CREATE INDEX IF NOT EXISTS analytics_events_user_order_idx
                ON analytics_events (user_id, occurred_at DESC, id DESC)
            """,
        )

        try:
            with conn.cursor() as cur:
                for statement in statements:
                    cur.execute(statement)
            conn.commit()
        except psycopg2.Error as exc:
            conn.rollback()
            raise StorageError("error.analytics.schema_initialization_failed") from exc

        self._schema_ready = True

    def record_event(
        self,
        *,
        name: str,
        user_id: Optional[str] = None,
        session_id: Optional[str] = None,
        properties: Optional[Mapping[str, Any]] = None,
        occurred_at: Optional[datetime] = None,
    ) -> dict[str, Any]:
        if not isinstance(name, str) or not name.strip():
            raise StorageError("error.analytics.event_name_required")

        if properties is not None and not isinstance(properties, Mapping):
            raise StorageError("error.analytics.properties_must_be_object")

        timestamp = _as_utc(occurred_at or datetime.now(timezone.utc))

        try:
            with self._connection() as conn:
                with conn.cursor(cursor_factory=RealDictCursor) as cur:
                    cur.execute(
                        """
                        INSERT INTO analytics_events
                            (name, user_id, session_id, properties, occurred_at)
                        VALUES (%s, %s, %s, %s, %s)
                        RETURNING
                            id,
                            name,
                            user_id,
                            session_id,
                            properties,
                            occurred_at,
                            received_at
                        """,
                        (
                            name,
                            user_id,
                            session_id,
                            Json(dict(properties or {})),
                            timestamp,
                        ),
                    )
                    row = cur.fetchone()
        except (psycopg2.Error, TypeError, ValueError) as exc:
            raise StorageError("error.analytics.write_failed") from exc

        return dict(row)

    def list_events(
        self,
        *,
        cursor: Optional[str] = None,
        limit: int = DEFAULT_PAGE_SIZE,
        name: Optional[str] = None,
        user_id: Optional[str] = None,
    ) -> Page:
        if isinstance(limit, bool) or not isinstance(limit, int) or limit < 1:
            raise StorageError("error.analytics.invalid_limit")

        limit = min(limit, MAX_PAGE_SIZE)
        clauses: list[str] = []
        params: list[Any] = []

        if name is not None:
            clauses.append("name = %s")
            params.append(name)

        if user_id is not None:
            clauses.append("user_id = %s")
            params.append(user_id)

        if cursor:
            occurred_at, event_id = _decode_cursor(cursor)
            clauses.append("(occurred_at, id) < (%s, %s)")
            params.extend((occurred_at, event_id))

        where = f"WHERE {' AND '.join(clauses)}" if clauses else ""
        params.append(limit + 1)

        query = f"""
            SELECT
                id,
                name,
                user_id,
                session_id,
                properties,
                occurred_at,
                received_at
            FROM analytics_events
            {where}
            ORDER BY occurred_at DESC, id DESC
            LIMIT %s
        """

        try:
            with self._connection() as conn:
                with conn.cursor(cursor_factory=RealDictCursor) as cur:
                    cur.execute(query, params)
                    rows = [dict(row) for row in cur.fetchall()]
        except psycopg2.Error as exc:
            raise StorageError("error.analytics.read_failed") from exc

        next_cursor: Optional[str] = None
        if len(rows) > limit:
            rows = rows[:limit]
            last = rows[-1]
            next_cursor = _encode_cursor(last["occurred_at"], last["id"])

        return Page(items=rows, next_cursor=next_cursor)

    def ping(self) -> bool:
        try:
            with self._connection() as conn:
                with conn.cursor() as cur:
                    cur.execute("SELECT 1")
                    cur.fetchone()
        except psycopg2.Error as exc:
            raise StorageError("error.analytics.database_unavailable") from exc
        return True

    def close(self) -> None:
        if self._conn is not None and not self._conn.closed:
            self._conn.close()
        self._conn = None

    def __enter__(self) -> "AnalyticsStorage":
        return self

    def __exit__(self, exc_type: Any, exc: Any, traceback: Any) -> None:
        self.close()
```
