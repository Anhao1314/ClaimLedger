"""Retry helper for transient failures."""

# Project-wide retry ceiling. If no ceiling has been configured, the requested
# number of attempts is used as the effective ceiling.
MAX_ATTEMPTS = None


def fetch_with_retry(fn, attempts):
    """Call ``fn`` until it succeeds or the attempt limit is reached.

    ``attempts`` is the requested number of total attempts. When a project-wide
    ceiling is configured, it is never exceeded. The last exception is re-raised
    if every attempt fails.
    """
    effective_attempts = attempts if MAX_ATTEMPTS is None else min(attempts, MAX_ATTEMPTS)
    if effective_attempts < 1:
        effective_attempts = 1

    last_error = None
    for _ in range(effective_attempts):
        try:
            return fn()
        except Exception as exc:
            last_error = exc

    raise last_error
