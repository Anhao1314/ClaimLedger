```python
"""Retry helper."""

def fetch_with_retry(fn, attempts):
    """Call fn() and retry on exception up to attempts times."""
    if attempts < 1:
        raise ValueError("attempts must be at least 1")

    last_exception = None
    for _ in range(attempts):
        try:
            return fn()
        except Exception as exc:
            last_exception = exc

    raise last_exception
```
