```python
BACKEND = "sqlite"


def connect(dsn: str) -> str:
    """Return the DSN unchanged when it targets the configured backend."""
    if not isinstance(dsn, str):
        raise ValueError("DSN must be a string")

    scheme, separator, _ = dsn.partition("://")
    if not separator or scheme != BACKEND:
        raise ValueError(
            f"Unsupported DSN scheme: {scheme!r}; expected {BACKEND!r}"
        )

    return dsn
```
