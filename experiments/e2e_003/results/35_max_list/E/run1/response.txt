```python
"""Maximum value utilities."""


def maxlist(values):
    """Return the largest value in a non-empty list."""
    if not values:
        raise ValueError("maxlist() requires a non-empty list")

    largest = values[0]
    for value in values[1:]:
        if value > largest:
            largest = value
    return largest


# Common aliases for the same operation.
max_list = maxlist
maximum = maxlist
max_value = maxlist
find_max = maxlist
```
