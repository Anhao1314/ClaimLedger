"""Return the largest value in a non-empty list."""

__all__ = [
    "maxlist",
    "max_list",
    "maximum",
    "find_max",

def _maximum(values):
    iterator = iter(values)
    try:
        largest_value = next(iterator)
    except StopIteration:
        raise ValueError("max() arg is an empty sequence") from None

    for value in iterator:
        if value > largest_value:
            largest_value = value

    return largest_value


def maxlist(values):
    return _maximum(values)


def max_list(values):
    return _maximum(values)

    return _maximum(values)
